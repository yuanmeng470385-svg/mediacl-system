# 数据库自动备份 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 MedAgent PostgreSQL 增加每日自动备份（ECS 本地 7 天 + 阿里云 OSS 30 天），带完整性校验与可验证的恢复脚本。

**Architecture:** 两个 Python 脚本（`scripts/backup_db.py` 备份+校验+上传+清理，`scripts/restore_db.py` 恢复到临时库+行数校验），全部经 `docker exec medagent-postgres` 操作数据库（prod 不映射公网端口），oss2 SDK 上传 OSS，ECS crontab 每天 02:00 触发。设计文档：`docs/superpowers/specs/2026-08-12-db-backup-design.md`。

**Tech Stack:** Python 3 + subprocess（docker exec）+ oss2 + argparse + cron（部署端）

## Global Constraints

- 备份格式：`pg_dump -Fc`（custom 格式，自带压缩）；备份账号 `medagent`（002 迁移降级后仍有全表 SELECT，不需要 superuser）
- 容器名 `medagent-postgres`（`BACKUP_PG_CONTAINER` 可覆盖）；prod 下数据库无公网端口 → **一律 `docker exec` 进容器执行 psql/pg_dump/pg_restore**，不用 TCP 连接
- **二进制安全**：pg_dump 输出用 `subprocess.run(..., stdout=open(path,'wb'))` 直写文件；psql/pg_restore 的 stdin 用 `subprocess.run(..., input=<bytes>)`——禁止 shell 重定向（Windows CRLF 会破坏 custom 格式 dump）
- 容器内执行需要密码：统一 `docker exec -e PGPASSWORD=<密码>`，密码从 `.env` 的 `DATABASE_URL` 解析（`postgresql://user:pass@...`），缺省 `medagent123`（dev compose 默认值）
- 恢复**默认恢复到临时库 `medagent_restore`**，绝不自动 drop/覆盖生产库 `medagent`；覆盖需显式 `--yes`
- 恢复一律 `pg_restore --no-owner --no-privileges`（audit_logs 的 owner 是 NOLOGIN 占位角色，新环境不存在）
- 失败留痕：`logs/backup.log` 追加日志 + `logs/backup-failed` 标记文件 + 非零退出码。告警渠道由部署文档指定（cron MAILTO / 阿里云 CloudMonitor 监控标记文件），**不假装调用不存在的 ARMS 事件 API**
- `.env` 不入库；新增配置位进 `.env.example`；`backups/` 目录进 `.gitignore`
- 提交遵循 conventional commits，末尾 `Co-Authored-By: Claude Fable 5 <noreply@anthropic.com>`

```python
#!/usr/bin/env python3
"""MedAgent 数据库备份：pg_dump -Fc 全量备份 + 完整性校验。

用法：
    python scripts/backup_db.py               # 备份 + 校验 + 上传 OSS + 过期清理
    python scripts/backup_db.py --local-only  # 只备份本地 + 校验（不上传 OSS）

配置（自动读取仓库根 .env，环境变量优先）：
    BACKUP_DIR                  本地保留目录（默认 ./backups）
    BACKUP_RETENTION_LOCAL_DAYS 本地保留天数（默认 7）
    BACKUP_RETENTION_OSS_DAYS   OSS 保留天数（默认 30）
    BACKUP_OSS_PREFIX           OSS 对象前缀（默认 backups/）
    BACKUP_PG_CONTAINER         数据库容器名（默认 medagent-postgres）
    OSS_ACCESS_KEY_ID / OSS_ACCESS_KEY_SECRET / OSS_BUCKET / OSS_ENDPOINT

依赖：oss2（pip install oss2），仅 OSS 上传/下载路径需要。
失败处理：logs/backup.log 追加日志 + logs/backup-failed 标记文件 + 非零退出码。
"""
import argparse
import datetime
import os
import re
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent

DEFAULT_CONFIG = {
    "BACKUP_DIR": "./backups",
    "BACKUP_RETENTION_LOCAL_DAYS": "7",
    "BACKUP_RETENTION_OSS_DAYS": "30",
    "BACKUP_OSS_PREFIX": "backups/",
    "BACKUP_PG_CONTAINER": "medagent-postgres",
}


def load_config() -> dict:
    """读取仓库根 .env（若存在且环境未设置则补充），返回配置 dict。"""
    cfg = dict(DEFAULT_CONFIG)
    env_file = ROOT / ".env"
    if env_file.exists():
        for line in env_file.read_text(encoding="utf-8").splitlines():
            line = line.strip()
            if not line or line.startswith("#") or "=" not in line:
                continue
            key, _, value = line.partition("=")
            cfg.setdefault(key.strip(), value.strip())
    for key in list(cfg):
        env_val = os.getenv(key)
        if env_val:
            cfg[key] = env_val
    return cfg


def pg_password(cfg: dict) -> str:
    """从 .env 的 DATABASE_URL 解析密码（缺省 dev compose 的 medagent123）。"""
    url = cfg.get("DATABASE_URL", "postgresql://medagent:medagent123@localhost:5432/medagent")
    m = re.search(r"postgres(?:ql)?://([^:]+):([^@]+)@", url)
    return m.group(2) if m else "medagent123"


def log(msg: str, cfg: dict) -> None:
    """打印 + 追加写 logs/backup.log。"""
    line = f"{datetime.datetime.now().isoformat(timespec='seconds')} {msg}"
    print(line)
    log_dir = ROOT / "logs"
    log_dir.mkdir(exist_ok=True)
    with open(log_dir / "backup.log", "a", encoding="utf-8") as f:
        f.write(line + "\n")


def _docker_exec(cfg: dict, args: list, input_bytes: bytes | None = None) -> subprocess.CompletedProcess:
    """docker exec 进 postgres 容器执行命令（带 PGPASSWORD）。"""
    cmd = ["docker", "exec", "-e", f"PGPASSWORD={pg_password(cfg)}", cfg["BACKUP_PG_CONTAINER"], *args]
    return subprocess.run(cmd, input=input_bytes, capture_output=True)


def run_pg_dump(cfg: dict) -> Path:
    """docker exec pg_dump -Fc 全量备份到 BACKUP_DIR。返回 dump 文件路径。"""
    dump_dir = Path(cfg["BACKUP_DIR"])
    dump_dir.mkdir(parents=True, exist_ok=True)
    stamp = datetime.datetime.now().strftime("%Y%m%d-%H%M")
    dump_path = dump_dir / f"medagent-{stamp}.dump"
    # 二进制输出直写文件（不用 shell 重定向，防 Windows CRLF 破坏 custom 格式）
    with open(dump_path, "wb") as out:
        cmd = ["docker", "exec", "-e", f"PGPASSWORD={pg_password(cfg)}", cfg["BACKUP_PG_CONTAINER"],
               "pg_dump", "-U", "medagent", "-Fc", "medagent"]
        proc = subprocess.run(cmd, stdout=out, stderr=subprocess.PIPE)
    if proc.returncode != 0:
        dump_path.unlink(missing_ok=True)
        raise RuntimeError(f"pg_dump 失败: {proc.stderr.decode('utf-8', 'replace')[:500]}")
    return dump_path


def validate_dump(dump_path: Path, cfg: dict) -> None:
    """pg_restore --list 校验 dump 可解析 + 文件非零。失败抛 RuntimeError。"""
    if dump_path.stat().st_size == 0:
        raise RuntimeError(f"dump 文件为空: {dump_path}")
    with open(dump_path, "rb") as f:
        cmd = ["docker", "exec", "-i", "-e", f"PGPASSWORD={pg_password(cfg)}", cfg["BACKUP_PG_CONTAINER"],
               "pg_restore", "--list"]
        proc = subprocess.run(cmd, stdin=f, stdout=subprocess.DEVNULL, stderr=subprocess.PIPE)
    if proc.returncode != 0:
        raise RuntimeError(f"dump 校验失败（pg_restore --list 非零）: {proc.stderr.decode('utf-8', 'replace')[:500]}")


def main() -> int:
    parser = argparse.ArgumentParser(description="MedAgent 数据库备份")
    parser.add_argument("--local-only", action="store_true", help="只做本地备份+校验，不上传 OSS（开发环境用）")
    args = parser.parse_args()
    cfg = load_config()
    try:
        dump_path = run_pg_dump(cfg)
        validate_dump(dump_path, cfg)
    except Exception as exc:
        log(f"备份失败: {exc}", cfg)
        (ROOT / "logs" / "backup-failed").touch()
        return 1
    log(f"备份完成: {dump_path.name}（--local-only={'是' if args.local_only else '否'}）", cfg)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
def _oss_bucket(cfg: dict):
    """构造 oss2.Bucket（复用 .env 的 OSS_* 凭证）。凭证缺失/oss2 未装抛 RuntimeError。"""
    try:
        import oss2
    except ImportError:
        raise RuntimeError("oss2 未安装，请执行: pip install oss2")
    ak = cfg.get("OSS_ACCESS_KEY_ID", "")
    sk = cfg.get("OSS_ACCESS_KEY_SECRET", "")
    bucket_name = cfg.get("OSS_BUCKET", "")
    endpoint = cfg.get("OSS_ENDPOINT", "")
    if not (ak and sk and bucket_name and endpoint):
        raise RuntimeError(
            "未配置 OSS_ACCESS_KEY_ID/OSS_ACCESS_KEY_SECRET/OSS_BUCKET/OSS_ENDPOINT（见 .env.example）"
        )
    return oss2.Bucket(oss2.Auth(ak, sk), endpoint, bucket_name)


def upload_oss(cfg: dict, dump_path: Path) -> None:
    """上传 dump 到 OSS_BUCKET 的 BACKUP_OSS_PREFIX，ACL 显式私有。"""
    bucket = _oss_bucket(cfg)
    key = f"{cfg['BACKUP_OSS_PREFIX']}{dump_path.name}"
    bucket.put_object_from_file(key, str(dump_path), headers={"x-oss-object-acl": "private"})
    log(f"OSS 上传完成: {bucket.bucket_name}/{key}", cfg)


def cleanup_local(cfg: dict) -> int:
    """删除 BACKUP_DIR 中超过 BACKUP_RETENTION_LOCAL_DAYS 天的备份，返回删除数。"""
    days = int(cfg["BACKUP_RETENTION_LOCAL_DAYS"])
    cutoff = datetime.datetime.now() - datetime.timedelta(days=days)
    pattern = re.compile(r"^medagent-(\d{8})-\d{4}\.dump$")
    removed = 0
    for p in Path(cfg["BACKUP_DIR"]).glob("medagent-*.dump"):
        m = pattern.match(p.name)
        if not m:
            continue
        file_date = datetime.datetime.strptime(m.group(1), "%Y%m%d")
        if file_date < cutoff:
            p.unlink(missing_ok=True)
            removed += 1
    if removed:
        log(f"本地清理: 删除 {removed} 个过期备份（>{days} 天）", cfg)
    return removed


def cleanup_oss(cfg: dict) -> int:
    """删除 OSS 中超过 BACKUP_RETENTION_OSS_DAYS 天的对象，返回删除数。"""
    try:
        import oss2
    except ImportError:
        raise RuntimeError("oss2 未安装，请执行: pip install oss2")
    days = int(cfg["BACKUP_RETENTION_OSS_DAYS"])
    cutoff = datetime.datetime.now() - datetime.timedelta(days=days)
    prefix = cfg["BACKUP_OSS_PREFIX"]
    bucket = _oss_bucket(cfg)
    pattern = re.compile(rf"^{re.escape(prefix)}medagent-(\d{{8}})-\d{{4}}\.dump$")
    removed = 0
    for obj in oss2.ObjectIterator(bucket, prefix=prefix):
        m = pattern.match(obj.key)
        if not m:
            continue
        obj_date = datetime.datetime.strptime(m.group(1), "%Y%m%d")
        if obj_date < cutoff:
            bucket.delete_object(obj.key)
            removed += 1
    if removed:
        log(f"OSS 清理: 删除 {removed} 个过期对象（>{days} 天）", cfg)
    return removed
```

```python
#!/usr/bin/env python3
"""MedAgent 数据库恢复：把备份 dump 恢复到临时库验证可恢复性（默认不碰生产库）。

用法：
    python scripts/restore_db.py --source local             # 本地最近一份
    python scripts/restore_db.py --source local --dump backups/medagent-20260812-0200.dump
    python scripts/restore_db.py --source oss               # 从 OSS 下载最近一份
    python scripts/restore_db.py --source local --target-db medagent_restore --yes

配置同 backup_db.py（复用 load_config）。
恢复默认到 medagent_restore 临时库，绝不自动覆盖生产库；覆盖需显式 --yes。
恢复生产库的步骤（人执行，勿自动化）：
    1) 停所有服务（docker compose -f deploy/docker-compose.prod.yml stop）
    2) docker exec medagent-postgres psql -U medagent -c "DROP DATABASE medagent WITH (FORCE)"
    3) python scripts/restore_db.py --source oss --target-db medagent --yes
    4) 重启服务；审计不可篡改配置（002 迁移）需在新库重新执行
"""
import argparse
import datetime
import re
import subprocess
import sys
from pathlib import Path

from backup_db import load_config, log, pg_password, _oss_bucket, ROOT

BACKUP_DIR = Path(load_config()["BACKUP_DIR"])


def _psql(cfg: dict, db: str, sql: str) -> subprocess.CompletedProcess:
    """docker exec psql 执行单条 SQL，返回 CompletedProcess。"""
    cmd = ["docker", "exec", "-e", f"PGPASSWORD={pg_password(cfg)}", cfg["BACKUP_PG_CONTAINER"],
           "psql", "-U", "medagent", "-d", db, "-t", "-A", "-c", sql]
    return subprocess.run(cmd, capture_output=True)


def pick_dump(cfg: dict, source: str, dump_arg: str | None) -> Path:
    """定位 dump 文件：本地按 --dump 或取最近一份；OSS 下载最近一份到 BACKUP_DIR/restore/。"""
    if source == "local":
        if dump_arg:
            p = Path(dump_arg)
            if not p.exists():
                raise RuntimeError(f"dump 文件不存在: {dump_arg}")
            return p
        dumps = sorted(BACKUP_DIR.glob("medagent-*.dump"))
        if not dumps:
            raise RuntimeError(f"{cfg['BACKUP_DIR']} 下没有备份文件")
        return dumps[-1]
    # OSS：对象 key 按字典序 = 时间序，取最近一份
    try:
        import oss2
    except ImportError:
        raise RuntimeError("oss2 未安装，请执行: pip install oss2")
    bucket = _oss_bucket(cfg)
    prefix = cfg["BACKUP_OSS_PREFIX"]
    pattern = re.compile(rf"^{re.escape(prefix)}medagent-(\d{{8}})-\d{{4}}\.dump$")
    objs = [o for o in oss2.ObjectIterator(bucket, prefix=prefix) if pattern.match(o.key)]
    if not objs:
        raise RuntimeError("OSS 上没有备份对象")
    latest = sorted(objs, key=lambda o: o.key)[-1]
    restore_dir = BACKUP_DIR / "restore"
    restore_dir.mkdir(parents=True, exist_ok=True)
    target = restore_dir / latest.key.rsplit("/", 1)[-1]
    bucket.get_object_to_file(latest.key, str(target))
    log(f"OSS 下载完成: {latest.key}", cfg)
    return target


def ensure_target_db(cfg: dict, target_db: str, yes: bool) -> None:
    """目标库已存在时要求确认（--yes 跳过）后 drop 重建，owner 为 medagent。"""
    proc = _psql(cfg, "postgres", f"SELECT 1 FROM pg_database WHERE datname='{target_db}'")
    exists = proc.stdout.decode("utf-8", "replace").strip() == "1"
    if exists:
        if not yes:
            answer = input(f"目标库 {target_db} 已存在，drop 重建？[y/N] ")
            if answer.strip().lower() != "y":
                raise RuntimeError("已取消")
        proc = _psql(cfg, "postgres", f"DROP DATABASE {target_db} WITH (FORCE)")
        if proc.returncode != 0:
            raise RuntimeError(f"drop 目标库失败: {proc.stderr.decode('utf-8', 'replace')[:300]}")
    proc = _psql(cfg, "postgres", f"CREATE DATABASE {target_db} OWNER medagent")
    if proc.returncode != 0:
        raise RuntimeError(f"创建目标库失败: {proc.stderr.decode('utf-8', 'replace')[:300]}")


def restore_dump(cfg: dict, dump_path: Path, target_db: str) -> None:
    """pg_restore --no-owner --no-privileges 灌入目标库（stdin 喂 dump 字节）。"""
    with open(dump_path, "rb") as f:
        cmd = ["docker", "exec", "-i", "-e", f"PGPASSWORD={pg_password(cfg)}", cfg["BACKUP_PG_CONTAINER"],
               "pg_restore", "--no-owner", "--no-privileges", "-d", target_db]
        proc = subprocess.run(cmd, stdin=f, capture_output=True)
    if proc.returncode != 0:
        raise RuntimeError(f"pg_restore 失败: {proc.stderr.decode('utf-8', 'replace')[:500]}")


def _query_counts(cfg: dict, db: str, tables: tuple) -> dict:
    """返回 {表名: 行数}。"""
    sql = " UNION ALL ".join(f"SELECT '{t}', COUNT(*) FROM {t}" for t in tables)
    proc = _psql(cfg, db, sql)
    if proc.returncode != 0:
        raise RuntimeError(f"查询 {db} 行数失败: {proc.stderr.decode('utf-8', 'replace')[:300]}")
    counts = {}
    for line in proc.stdout.decode("utf-8", "replace").splitlines():
        if "|" in line:
            t, _, n = line.partition("|")
            counts[t.strip()] = int(n.strip())
    return counts


def verify_counts(cfg: dict, src_db: str, target_db: str,
                  tables: tuple = ("users", "conversations", "audit_logs")) -> dict:
    """对比源库与目标库行数，返回 (src, tgt, mismatches)。"""
    src = _query_counts(cfg, src_db, tables)
    tgt = _query_counts(cfg, target_db, tables)
    mismatches = {t: (src.get(t), tgt.get(t)) for t in tables if src.get(t) != tgt.get(t)}
    return {"source": src, "target": tgt, "mismatches": mismatches}


def main() -> int:
    parser = argparse.ArgumentParser(description="MedAgent 数据库恢复（默认恢复临时库）")
    parser.add_argument("--source", choices=["local", "oss"], default="local", help="备份来源")
    parser.add_argument("--dump", default=None, help="指定 dump 文件（默认取最近一份）")
    parser.add_argument("--target-db", default="medagent_restore", help="恢复目标库（默认临时库）")
    parser.add_argument("--yes", action="store_true", help="目标库已存在时跳过确认直接重建")
    args = parser.parse_args()
    cfg = load_config()
    try:
        dump_path = pick_dump(cfg, args.source, args.dump)
        ensure_target_db(cfg, args.target_db, args.yes)
        restore_dump(cfg, dump_path, args.target_db)
        result = verify_counts(cfg, "medagent", args.target_db)
    except Exception as exc:
        log(f"恢复失败: {exc}", cfg)
        return 1
    print(f"恢复完成: {dump_path} → {args.target_db}")
    print(f"源库行数: {result['source']}")
    print(f"目标库行数: {result['target']}")
    if result["mismatches"]:
        log(f"恢复校验不一致: {result['mismatches']}", cfg)
        return 1
    log(f"恢复验证通过: {dump_path.name} → {args.target_db} 行数一致", cfg)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
